<script type="module">
  // 1. IMPORTAR FIREBASE V9
  import { initializeApp } from "https://www.gstatic.com/firebasejs/10.8.1/firebase-app.js";
  import { getAuth, signInWithEmailAndPassword, createUserWithEmailAndPassword, onAuthStateChanged, signOut } from "https://www.gstatic.com/firebasejs/10.8.1/firebase-auth.js";
  import { getFirestore, doc, setDoc, getDoc, collection, getDocs, onSnapshot } from "https://www.gstatic.com/firebasejs/10.8.1/firebase-firestore.js";

  // ==========================================
  // 🔴 CONFIGURACIÓN (REEMPLAZAR AQUÍ)
  // ==========================================
  const ADMIN_EMAIL = "alejandr92009@hotmail.com"; // <-- Pon tu correo de entrenador aquí
  
  const firebaseConfig = {
  apiKey: "AIzaSyAaod77oOcmeZKAJ0lvR1s5PDq23ob6CFI",
  authDomain: "forgelifting-636d9.firebaseapp.com",
  projectId: "forgelifting-636d9",
  storageBucket: "forgelifting-636d9.firebasestorage.app",
  messagingSenderId: "476741821266",
  appId: "1:476741821266:web:f90ce2ecb43f833b1fd08b"
};// ==========================================

  const app = initializeApp(firebaseConfig);
  const auth = getAuth(app);
  const db = getFirestore(app);

  // Variables Globales
  let currentUser = null;
  let isCoach = false;
  let currentViewedAthleteId = null; // UID del atleta que se está viendo
  let routineState = {}; 
  let currentWeek = 1;
  let saveTimeout = null;

  // UI Elements
  const authScreen = document.getElementById('auth-screen');
  const appScreen = document.getElementById('app-screen');
  const authError = document.getElementById('authError');
  const coachPanel = document.getElementById('coachPanel');
  const athleteSelect = document.getElementById('athleteSelect');
  const weekContent = document.getElementById('weekContent');
  const saveIndicator = document.getElementById('saveIndicator');

  // --- Iconos SVG (Igual que antes) ---
  const ICON_SQUAT = `<svg viewBox="0 0 48 48" fill="none" xmlns="http://www.w3.org/2000/svg"><line x1="10" y1="8" x2="10" y2="40" stroke="currentColor" stroke-width="3"/><line x1="38" y1="8" x2="38" y2="40" stroke="currentColor" stroke-width="3"/><line x1="3" y1="17" x2="45" y2="17" stroke="currentColor" stroke-width="4"/><circle cx="7" cy="17" r="5" fill="currentColor"/><circle cx="41" cy="17" r="5" fill="currentColor"/></svg>`;
  const ICON_BENCH = `<svg viewBox="0 0 48 48" fill="none" xmlns="http://www.w3.org/2000/svg"><rect x="13" y="29" width="22" height="6" rx="1.5" fill="currentColor"/><line x1="15" y1="35" x2="15" y2="41" stroke="currentColor" stroke-width="3"/><line x1="33" y1="35" x2="33" y2="41" stroke="currentColor" stroke-width="3"/><line x1="3" y1="15" x2="45" y2="15" stroke="currentColor" stroke-width="4"/><circle cx="7" cy="15" r="5" fill="currentColor"/><circle cx="41" cy="15" r="5" fill="currentColor"/></svg>`;
  const ICON_DEADLIFT = `<svg viewBox="0 0 48 48" fill="none" xmlns="http://www.w3.org/2000/svg"><line x1="3" y1="24" x2="45" y2="24" stroke="currentColor" stroke-width="4"/><circle cx="7" cy="24" r="6.5" fill="currentColor"/><circle cx="41" cy="24" r="6.5" fill="currentColor"/></svg>`;

  const LIFTS = [
    { key:'sentadilla', label:'Sentadilla', color:'blue', icon:ICON_SQUAT },
    { key:'pressBanca', label:'Press Banca', color:'yellow', icon:ICON_BENCH },
    { key:'pesoMuerto', label:'Peso Muerto', color:'red', icon:ICON_DEADLIFT }
  ];

  // --- Lógica de Base de Datos Base ---
  function emptyLift(){ return { series:'', reps:'', carga:'', equipamiento:'', metodo:'', sugerencia:'' }; }
  function emptyWeek(){ return { sentadilla:emptyLift(), pressBanca:emptyLift(), pesoMuerto:emptyLift(), feedback:{ valoracion:0, comentario:'' } }; }
  function generateEmptyRoutine(){ 
    const rut = {}; 
    for(let w=1; w<=8; w++) rut[w] = emptyWeek(); 
    return rut; 
  }

  // --- Autenticación ---
  document.getElementById('btnLogin').addEventListener('click', async () => {
    const email = document.getElementById('emailInput').value;
    const pass = document.getElementById('passwordInput').value;
    try { await signInWithEmailAndPassword(auth, email, pass); authError.style.display='none'; } 
    catch(err) { authError.innerText = "Error: Verifica tus credenciales."; authError.style.display='block'; }
  });

  document.getElementById('btnRegister').addEventListener('click', async () => {
    const email = document.getElementById('emailInput').value;
    const pass = document.getElementById('passwordInput').value;
    try {
      const userCred = await createUserWithEmailAndPassword(auth, email, pass);
      // Guardar al atleta en la lista de usuarios de Firestore
      await setDoc(doc(db, "users", userCred.user.uid), { email: email, role: "athlete" });
      authError.style.display='none';
    } catch(err) { authError.innerText = "Error al crear cuenta."; authError.style.display='block'; }
  });

  document.getElementById('btnLogout').addEventListener('click', () => signOut(auth));

  // Escuchar estado de sesión
  onAuthStateChanged(auth, async (user) => {
    if (user) {
      currentUser = user;
      document.getElementById('userEmailDisplay').innerText = user.email;
      isCoach = (user.email === ADMIN_EMAIL);
      authScreen.classList.add('hidden');
      appScreen.classList.remove('hidden');

      if (isCoach) {
        coachPanel.style.display = 'flex';
        loadAthletesList();
        weekContent.innerHTML = '<p style="text-align:center;">Selecciona un atleta arriba para editar su rutina.</p>';
      } else {
        coachPanel.style.display = 'none';
        currentViewedAthleteId = user.uid; // El atleta se ve a sí mismo
        await fetchRoutine(user.uid);
      }
    } else {
      currentUser = null;
      authScreen.classList.remove('hidden');
      appScreen.classList.add('hidden');
    }
  });

  // --- Vistas Coach ---
  async function loadAthletesList() {
    const querySnapshot = await getDocs(collection(db, "users"));
    athleteSelect.innerHTML = '<option value="">Selecciona un atleta...</option>';
    querySnapshot.forEach((doc) => {
      if(doc.data().role === 'athlete') {
        const opt = document.createElement('option');
        opt.value = doc.id; opt.innerText = doc.data().email;
        athleteSelect.appendChild(opt);
      }
    });
  }

  athleteSelect.addEventListener('change', async (e) => {
    currentViewedAthleteId = e.target.value;
    if(currentViewedAthleteId) await fetchRoutine(currentViewedAthleteId);
    else weekContent.innerHTML = '';
  });

  // --- Recuperar y Guardar Rutinas (Firestore) ---
  async function fetchRoutine(uid) {
    weekContent.innerHTML = '<p>Cargando datos desde la nube...</p>';
    const docRef = doc(db, "routines", uid);
    const docSnap = await getDoc(docRef);
    
    if (docSnap.exists()) {
      routineState = docSnap.data();
    } else {
      routineState = generateEmptyRoutine();
      if(isCoach) await setDoc(docRef, routineState); // Solo el coach inicializa vacías
    }
    renderTabs();
    renderWeek(currentWeek);
  }

  async function triggerSave() {
    if(!currentViewedAthleteId) return;
    saveIndicator.innerText = "Guardando..."; saveIndicator.className = "save-indicator pending";
    clearTimeout(saveTimeout);
    
    saveTimeout = setTimeout(async () => {
      try {
        await setDoc(doc(db, "routines", currentViewedAthleteId), routineState);
        saveIndicator.innerText = "Guardado ✓"; saveIndicator.className = "save-indicator ok";
        setTimeout(() => { saveIndicator.innerText = ""; }, 2000);
      } catch(e) { saveIndicator.innerText = "Error"; saveIndicator.style.color = "red"; }
    }, 1000); // Guarda 1 segundo después de dejar de escribir
  }

  // --- Renderizado de UI ---
  function renderTabs() {
    let html='';
    for(let w=1; w<=8; w++){
      html+=`<button type="button" class="week-tab ${w===currentWeek?'active':''}" data-week="${w}">
        <span class="week-tab-ring"><span class="week-tab-num">${w}</span></span>
      </button>`;
    }
    document.getElementById('weekTabs').innerHTML = html;
  }

  document.getElementById('weekTabs').addEventListener('click', (e) => {
    const btn = e.target.closest('.week-tab');
    if(!btn) return;
    currentWeek = parseInt(btn.dataset.week, 10);
    renderTabs();
    renderWeek(currentWeek);
  });

  function renderWeek(w) {
    if(!routineState[w]) return;
    
    // Si NO es coach, los campos de entreno son readonly, el feedback sí se puede editar.
    const roAttr = isCoach ? '' : 'readonly';
    const disableBtn = isCoach ? '' : 'disabled';

    const liftsHTML = LIFTS.map(lift => {
      const data = routineState[w][lift.key];
      return `
      <section class="lift-card lift-${lift.color}">
        <div class="lift-card-header"><span class="lift-icon">${lift.icon}</span><h2 class="lift-title">${lift.label}</h2></div>
        <div class="field-group">
          <div class="field"><label>Series</label><input type="text" class="mono-input" value="${data.series||''}" data-lift="${lift.key}" data-field="series" ${roAttr}></div>
          <div class="field"><label>Reps</label><input type="text" class="mono-input" value="${data.reps||''}" data-lift="${lift.key}" data-field="reps" ${roAttr}></div>
          <div class="field"><label>Carga</label><input type="text" class="mono-input" value="${data.carga||''}" data-lift="${lift.key}" data-field="carga" ${roAttr}></div>
        </div>
        <div class="field"><label>Notas del Entrenador</label><textarea rows="2" data-lift="${lift.key}" data-field="sugerencia" ${roAttr}>${data.sugerencia||''}</textarea></div>
      </section>`;
    }).join('');

    const fb = routineState[w].feedback;
    let plates='';
    // El atleta SIEMPRE puede editar el feedback. El entrenador puede verlo.
    for(let i=1;i<=5;i++){ plates+=`<button type="button" class="plate-btn ${i<=fb.valoracion?'filled':''}" data-val="${i}"></button>`; }
    
    const fbHTML = `
    <section class="feedback-card">
      <h2 style="color:var(--chalk); font-family:'Bebas Neue',sans-serif; font-size:22px; margin-top:0;">Feedback del Atleta (Semana ${w})</h2>
      <div class="rating-row"><span style="font-size:12px; color:var(--chalk-dust); text-transform:uppercase;">Sensación</span><div class="plate-rating">${plates}</div></div>
      <div class="field"><textarea rows="2" placeholder="Notas de fatiga, dolores, técnica..." id="fbComment">${fb.comentario||''}</textarea></div>
    </section>`;

    weekContent.innerHTML = `<div class="lifts-grid">${liftsHTML}</div>${fbHTML}`;
  }

  // --- Captura de inputs ---
  weekContent.addEventListener('input', (e) => {
    const t = e.target;
    // Si cambia un input de rutina (solo si es coach o no tiene readonly)
    if(t.dataset.lift && !t.hasAttribute('readonly')) {
      routineState[currentWeek][t.dataset.lift][t.dataset.field] = t.value;
      triggerSave();
    }
    // Si cambia el comentario de feedback (Cualquiera puede, idealmente el atleta)
    if(t.id === 'fbComment') {
      routineState[currentWeek].feedback.comentario = t.value;
      triggerSave();
    }
  });

  weekContent.addEventListener('click', (e) => {
    const btn = e.target.closest('.plate-btn');
    if(!btn) return;
    // Seleccionador de estrellas
    const val = parseInt(btn.dataset.val, 10);
    routineState[currentWeek].feedback.valoracion = val;
    renderWeek(currentWeek); // Re-dibuja para iluminar las estrellas
    triggerSave();
  });

</script>
