  window.calcularCoeficientes = function() {
    const gender = document.getElementById('calc-gender').value;
    const bw = parseFloat(document.getElementById('calc-bw').value);
    const total = parseFloat(document.getElementById('calc-total').value);

    if(!bw || !total || bw <= 0 || total <= 0) return;

    let dots = 0;
    if (gender === 'M') {
      const denom = -307.5820 + (2.0558 * bw) + (-0.00228492 * Math.pow(bw,2)) + (-0.0000305886 * Math.pow(bw,3)) + (0.00000028212 * Math.pow(bw,4));
      dots = (total * 500) / denom;
    } else {
      const denom = -57.962880 + (1.363949 * bw) + (-0.003362143 * Math.pow(bw,2)) + (0.0000185816 * Math.pow(bw,3)) + (-0.000000034509 * Math.pow(bw,4));
      dots = (total * 500) / denom;
    }

    let gl = 0;
    if (gender === 'M') {
      gl = (total * 100) / (1199.72839 - 1025.18162 * Math.exp(-0.00921 * bw));
    } else {
      gl = (total * 100) / (610.32796 - 1045.59282 * Math.exp(-0.03048 * bw));
    }

    let wilks = 0;
    if (gender === 'M') {
      const denom = -216.0475144 + 16.2606339*bw - 0.002388645*Math.pow(bw,2) - 0.00113732*Math.pow(bw,3) + 7.01863E-06*Math.pow(bw,4) - 1.291E-08*Math.pow(bw,5);
      wilks = total * 500 / denom;
    } else {
      const denom = 594.31747775582 - 27.23842536447*bw + 0.82112226871*Math.pow(bw,2) - 0.00930733913*Math.pow(bw,3) + 0.00004731582*Math.pow(bw,4) - 0.00000009054*Math.pow(bw,5);
      wilks = total * 500 / denom;
    }

    document.getElementById('res-dots').innerText = dots.toFixed(2);
    document.getElementById('res-gl').innerText = gl.toFixed(2);
    document.getElementById('res-wilks').innerText = wilks.toFixed(2);
  };

  const ICON_SQUAT = `<svg viewBox="0 0 48 48" fill="none" xmlns="http://www.w3.org/2000/svg"><line x1="10" y1="8" x2="10" y2="40" stroke="currentColor" stroke-width="3" stroke-linecap="round"/><line x1="38" y1="8" x2="38" y2="40" stroke="currentColor" stroke-width="3" stroke-linecap="round"/><line x1="3" y1="17" x2="45" y2="17" stroke="currentColor" stroke-width="4" stroke-linecap="round"/><circle cx="7" cy="17" r="5" fill="currentColor"/><circle cx="41" cy="17" r="5" fill="currentColor"/></svg>`;
  const ICON_BENCH = `<svg viewBox="0 0 48 48" fill="none" xmlns="http://www.w3.org/2000/svg"><rect x="13" y="29" width="22" height="6" rx="1.5" fill="currentColor"/><line x1="15" y1="35" x2="15" y2="41" stroke="currentColor" stroke-width="3" stroke-linecap="round"/><line x1="33" y1="35" x2="33" y2="41" stroke="currentColor" stroke-width="3" stroke-linecap="round"/><line x1="3" y1="15" x2="45" y2="15" stroke="currentColor" stroke-width="4" stroke-linecap="round"/><circle cx="7" cy="15" r="5" fill="currentColor"/><circle cx="41" cy="15" r="5" fill="currentColor"/></svg>`;
  const ICON_DEADLIFT = `<svg viewBox="0 0 48 48" fill="none" xmlns="http://www.w3.org/2000/svg"><line x1="3" y1="24" x2="45" y2="24" stroke="currentColor" stroke-width="4" stroke-linecap="round"/><circle cx="7" cy="24" r="6.5" fill="currentColor"/><circle cx="41" cy="24" r="6.5" fill="currentColor"/><line x1="3" y1="38" x2="45" y2="38" stroke="currentColor" stroke-width="2" opacity="0.45" stroke-linecap="round"/></svg>`;

  const LIFTS = [
    { key:'sentadilla', label:'Sentadilla', color:'blue', icon:ICON_SQUAT, accDatalist:'accSentadillaOptions' },
    { key:'pressBanca', label:'Press Banca', color:'yellow', icon:ICON_BENCH, accDatalist:'accBancaOptions' },
    { key:'pesoMuerto', label:'Peso Muerto', color:'red', icon:ICON_DEADLIFT, accDatalist:'accPesoMuertoOptions' }
  ];

  const STORAGE_KEY = 'forgelifting-v1';
  let state = null;

  function emptyLift(){ return { series:'', reps:'', carga:'', equipamiento:'', metodo:'', descanso:'', sugerencia:'', accesorios:[] }; }
  function emptyWeek(){ return { sentadilla:emptyLift(), pressBanca:emptyLift(), pesoMuerto:emptyLift(), feedback:{ valoracion:0, comentario:'' } }; }
  function defaultState(){
    const semanas={};
    for(let w=1; w<=10; w++){ semanas[w]=emptyWeek(); }
    return { duracion:8, semanaActiva:1, semanas };
  }
  function clamp(n,min,max){ return Math.max(min, Math.min(max, n)); }
  function genId(){ return 'a'+Date.now().toString(36)+Math.random().toString(36).slice(2,7); }

  function mergeLift(l){
    const base=emptyLift();
    if(!l) return base;
    base.series=l.series||''; base.reps=l.reps||''; base.carga=l.carga||'';
    base.equipamiento=l.equipamiento||''; base.metodo=l.metodo||''; base.descanso=l.descanso||'';
    base.sugerencia=l.sugerencia||'';
    base.accesorios = Array.isArray(l.accesorios) ? l.accesorios.map(a=>({
      id:a.id||genId(), nombre:a.nombre||'', series:a.series||'', reps:a.reps||'', carga:a.carga||''
    })) : [];
    return base;
  }
  function mergeWithDefaults(parsed){
    const base=defaultState();
    if(!parsed || typeof parsed!=='object') return base;
    base.duracion = clamp(parseInt(parsed.duracion,10)||8,1,10);
    if(parsed.semanas){
      for(let w=1; w<=10; w++){
        const pw = parsed.semanas[w] || parsed.semanas[String(w)];
        if(pw){
          base.semanas[w]={
            sentadilla:mergeLift(pw.sentadilla),
            pressBanca:mergeLift(pw.pressBanca),
            pesoMuerto:mergeLift(pw.pesoMuerto),
            feedback:{ valoracion: clamp(parseInt(pw.feedback && pw.feedback.valoracion,10)||0,0,5), comentario: (pw.feedback && pw.feedback.comentario) || '' }
          };
        }
      }
    }
    base.semanaActiva = clamp(parseInt(parsed.semanaActiva,10)||1,1,base.duracion);
    return base;
  }

  function escapeAttr(str){ return String(str||'').replace(/&/g,'&amp;').replace(/"/g,'&quot;').replace(/</g,'&lt;').replace(/>/g,'&gt;'); }
  function escapeHTML(str){ return escapeAttr(str); }

  function weekCompletion(w){
    const week = state.semanas[w];
    let filled=0; const total = LIFTS.length*3;
    LIFTS.forEach(l=>{
      const d=week[l.key];
      if(d.series.trim()) filled++; if(d.reps.trim()) filled++; if(d.carga.trim()) filled++;
    });
    return Math.round((filled/total)*100);
  }

  const appEl = document.getElementById('app');
  const weekTabsEl = document.getElementById('weekTabs');
  const weekContentEl = document.getElementById('weekContent');
  const legendEl = document.getElementById('legend');
  const duracionSelect = document.getElementById('duracionSelect');
  const saveIndicatorEl = document.getElementById('saveIndicator');
  const modalEl = document.getElementById('confirmModal');

  function renderLegend(){ legendEl.innerHTML = LIFTS.map(l=>`<span class="legend-item legend-${l.color}">${l.icon}${l.label}</span>`).join(''); }
  function renderDuracionOptions(){
    let html='';
    for(let i=1;i<=10;i++){ html+=`<option value="${i}" ${i===state.duracion?'selected':''}>${i} semana${i>1?'s':''}</option>`; }
    duracionSelect.innerHTML=html;
  }

  function renderTabs(){
    let html='';
    for(let w=1; w<=state.duracion; w++){
      const pct=weekCompletion(w);
      html+=`<button type="button" class="week-tab ${w===state.semanaActiva?'active':''}" data-action="switch-week" data-week="${w}" aria-current="${w===state.semanaActiva?'true':'false'}">
        <span class="week-tab-ring" style="--pct:${pct}"><span class="week-tab-num">${w}</span></span>
      </button>`;
    }
    weekTabsEl.innerHTML = html;
  }
  function updateTabProgress(w){
    const btn = weekTabsEl.querySelector(`.week-tab[data-week="${w}"] .week-tab-ring`);
    if(btn) btn.style.setProperty('--pct', weekCompletion(w));
  }

  function liftCardTemplate(weekNum, lift){
    const data = state.semanas[weekNum][lift.key];
    const accessoriesHTML = data.accesorios.map(acc=>accessoryRowTemplate(weekNum, lift, acc)).join('');
    return `
    <section class="lift-card lift-${lift.color}">
      <div class="lift-card-header"><span class="lift-icon">${lift.icon}</span><h2 class="lift-title">${lift.label}</h2></div>
      <div class="field-group">
        <div class="field"><label>Series</label><input type="text" class="mono-input" placeholder="4" value="${escapeAttr(data.series)}" data-action="lift-field" data-week="${weekNum}" data-lift="${lift.key}" data-field="series"></div>
        <div class="field"><label>Repeticiones</label><input type="text" class="mono-input" placeholder="5" value="${escapeAttr(data.reps)}" data-action="lift-field" data-week="${weekNum}" data-lift="${lift.key}" data-field="reps"></div>
        <div class="field"><label>Carga</label><input type="text" class="mono-input" placeholder="80% 1RM" value="${escapeAttr(data.carga)}" data-action="lift-field" data-week="${weekNum}" data-lift="${lift.key}" data-field="carga"></div>
      </div>
      <div class="field-group">
        <div class="field"><label>Equipamiento</label><input type="text" list="equipoOptions" placeholder="Cinturón..." value="${escapeAttr(data.equipamiento)}" data-action="lift-field" data-week="${weekNum}" data-lift="${lift.key}" data-field="equipamiento"></div>
        <div class="field"><label>Método</label><input type="text" list="metodoOptions" placeholder="RPE..." value="${escapeAttr(data.metodo)}" data-action="lift-field" data-week="${weekNum}" data-lift="${lift.key}" data-field="metodo"></div>
        <div class="field"><label>Descanso</label><input type="text" class="mono-input" placeholder="3-5 min" value="${escapeAttr(data.descanso)}" data-action="lift-field" data-week="${weekNum}" data-lift="${lift.key}" data-field="descanso"></div>
      </div>
      <div class="field field-suggestion"><label>Sugerencia del entrenador</label><textarea rows="2" placeholder="Foco técnico de la semana, notas..." data-action="lift-field" data-week="${weekNum}" data-lift="${lift.key}" data-field="sugerencia">${escapeHTML(data.sugerencia)}</textarea></div>
      <div class="accessories-block">
        <div class="accessories-header"><h3>Accesorios</h3><button type="button" class="btn-add" data-action="add-accessory" data-week="${weekNum}" data-lift="${lift.key}">+ Añadir</button></div>
        <div class="accessories-list" id="accessories-${weekNum}-${lift.key}">
          ${data.accesorios.length ? `<div class="accessories-columns"><span>Ejercicio</span><span>Series</span><span>Reps</span><span>Carga</span><span></span></div>${accessoriesHTML}` : '<p class="empty-hint">Sin accesorios añadidos</p>'}
        </div>
      </div>
    </section>`;
  }

  function accessoryRowTemplate(weekNum, lift, acc){
    return `<div class="accessory-row">
      <input type="text" list="${lift.accDatalist}" placeholder="Ejercicio" value="${escapeAttr(acc.nombre)}" data-action="accessory-field" data-week="${weekNum}" data-lift="${lift.key}" data-acc-id="${acc.id}" data-field="nombre">
      <input type="text" class="mono-input" placeholder="Series" value="${escapeAttr(acc.series)}" data-action="accessory-field" data-week="${weekNum}" data-lift="${lift.key}" data-acc-id="${acc.id}" data-field="series">
      <input type="text" class="mono-input" placeholder="Reps" value="${escapeAttr(acc.reps)}" data-action="accessory-field" data-week="${weekNum}" data-lift="${lift.key}" data-acc-id="${acc.id}" data-field="reps">
      <input type="text" class="mono-input" placeholder="Carga" value="${escapeAttr(acc.carga)}" data-action="accessory-field" data-week="${weekNum}" data-lift="${lift.key}" data-acc-id="${acc.id}" data-field="carga">
      <button type="button" class="btn-remove" aria-label="Eliminar accesorio" data-action="remove-accessory" data-week="${weekNum}" data-lift="${lift.key}" data-acc-id="${acc.id}">×</button>
    </div>`;
  }

  function renderAccessoriesList(weekNum, liftKey){
    const lift = LIFTS.find(l=>l.key===liftKey);
    const container = document.getElementById(`accessories-${weekNum}-${liftKey}`);
    if(!container) return;
    const arr = state.semanas[weekNum][liftKey].accesorios;
    if(!arr.length){ container.innerHTML = '<p class="empty-hint">Sin accesorios añadidos</p>'; return; }
    container.innerHTML = `<div class="accessories-columns"><span>Ejercicio</span><span>Series</span><span>Reps</span><span>Carga</span><span></span></div>` + arr.map(acc=>accessoryRowTemplate(weekNum, lift, acc)).join('');
  }

  function feedbackCardTemplate(weekNum){
    const fb = state.semanas[weekNum].feedback;
    let plates='';
    for(let i=1;i<=5;i++){ plates+=`<button type="button" class="plate-btn ${i<=fb.valoracion?'filled':''}" data-action="feedback-rating" data-week="${weekNum}" data-value="${i}"></button>`; }
    return `<section class="feedback-card"><h2>Feedback del atleta — Semana ${weekNum}</h2>
      <p class="feedback-sub">Cómo sentiste esta semana.</p>
      <div class="rating-row"><span class="rating-label">Sensación general</span><div class="plate-rating">${plates}</div></div>
      <div class="field"><label>Comentario</label><textarea rows="3" placeholder="Técnica, fatiga..." data-action="feedback-comentario" data-week="${weekNum}">${escapeHTML(fb.comentario)}</textarea></div>
    </section>`;
  }

  function renderWeekContent(weekNum){
    const liftsHTML = LIFTS.map(l=>liftCardTemplate(weekNum, l)).join('');
    weekContentEl.innerHTML = `<span class="week-heading-eyebrow">Semana ${weekNum} de ${state.duracion}</span><div class="lifts-grid">${liftsHTML}</div>${feedbackCardTemplate(weekNum)}`;
  }

  let saveTimeout=null;
  function showSaveState(text, cls){ saveIndicatorEl.textContent = text; saveIndicatorEl.className = 'save-indicator ' + cls; }
  function scheduleSave(){
    showSaveState('Guardando…','pending');
    clearTimeout(saveTimeout);
    saveTimeout = setTimeout(()=>{
      try{
        localStorage.setItem(STORAGE_KEY, JSON.stringify(state));
        showSaveState('Guardado ✓','ok');
        setTimeout(()=>{ if(saveIndicatorEl.textContent==='Guardado ✓') showSaveState('',''); }, 2200);
      }catch(e){ showSaveState('Error','err'); }
    }, 650);
  }

  function loadState(){
    try{
      const val = localStorage.getItem(STORAGE_KEY);
      state = val ? mergeWithDefaults(JSON.parse(val)) : defaultState();
    }catch(e){ state = defaultState(); }
  }

  function doReset(){ state = defaultState(); renderDuracionOptions(); renderTabs(); renderWeekContent(state.semanaActiva); scheduleSave(); }
  
  appEl.addEventListener('input', (e)=>{
    const t = e.target; const action = t.dataset.action; if(!action) return;
    if(action==='lift-field'){ state.semanas[t.dataset.week][t.dataset.lift][t.dataset.field] = t.value; updateTabProgress(t.dataset.week); scheduleSave(); }
    else if(action==='accessory-field'){ const acc = state.semanas[t.dataset.week][t.dataset.lift].accesorios.find(a=>a.id===t.dataset.accId); if(acc){ acc[t.dataset.field]=t.value; scheduleSave(); } }
    else if(action==='feedback-comentario'){ state.semanas[t.dataset.week].feedback.comentario = t.value; scheduleSave(); }
  });

  appEl.addEventListener('click', (e)=>{
    const btn = e.target.closest('[data-action]'); if(!btn) return; const action = btn.dataset.action;
    if(action==='switch-week'){ state.semanaActiva = parseInt(btn.dataset.week,10); renderTabs(); renderWeekContent(state.semanaActiva); }
    else if(action==='add-accessory'){ state.semanas[btn.dataset.week][btn.dataset.lift].accesorios.push({id:genId(), nombre:'', series:'', reps:'', carga:''}); renderAccessoriesList(btn.dataset.week, btn.dataset.lift); scheduleSave(); }
    else if(action==='remove-accessory'){ const arr = state.semanas[btn.dataset.week][btn.dataset.lift].accesorios; const idx = arr.findIndex(a=>a.id===btn.dataset.accId); if(idx>-1) arr.splice(idx,1); renderAccessoriesList(btn.dataset.week, btn.dataset.lift); scheduleSave(); }
    else if(action==='feedback-rating'){ const w = btn.dataset.week, val = parseInt(btn.dataset.value,10); state.semanas[w].feedback.valoracion = val; document.querySelectorAll(`[data-action="feedback-rating"][data-week="${w}"]`).forEach(b=>{ b.classList.toggle('filled', parseInt(b.dataset.value,10)<=val); }); scheduleSave(); }
    else if(action==='open-reset') modalEl.classList.remove('hidden');
    else if(action==='confirm-reset'){ doReset(); modalEl.classList.add('hidden'); }
    else if(action==='cancel-reset') modalEl.classList.add('hidden');
  });

  duracionSelect.addEventListener('change', (e)=>{
    state.duracion = clamp(parseInt(e.target.value,10)||8,1,10);
    if(state.semanaActiva > state.duracion) state.semanaActiva = state.duracion;
    renderTabs(); renderWeekContent(state.semanaActiva); scheduleSave();
  });

  function init(){
    renderLegend(); loadState(); renderDuracionOptions(); renderTabs(); renderWeekContent(state.semanaActiva);
  }
  init();
})();
